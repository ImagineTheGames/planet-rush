    class Component extends DCLogic {
      state = { data: null };
    
      componentDidMount() {
        const tick = async () => {
          try {
            const data = await (await fetch('/api/art')).json();
            // Re-render reloads every embedded board iframe — only do it on real change.
            const key = JSON.stringify(data);
            if (key !== this.__lastKey) { this.__lastKey = key; this.setState({ data }); }
          }
          catch { /* keep last */ }
        };
        tick();
        this.__poll = setInterval(tick, 30000);
      }
      componentWillUnmount() { clearInterval(this.__poll); }
    
      async choose(board, verdict, reason) {
        try {
          await fetch('/api/art/choice', { method: 'POST', body: JSON.stringify({ board, verdict, reason }) });
          this.setState({ data: await (await fetch('/api/art')).json() });
        } catch { /* poll retries */ }
      }
    
      renderVals() {
        const d = this.state.data ?? { boards: [], choices: {}, review: {}, work: {} };
        const review = d.review ?? {};
        const choices = d.choices ?? {};
        const work = d.work ?? {};
        const reviews = [];
        const refs = [];
        // A decision the developer MADE, and what the studio did about it. Picking a
        // concept used to move the card to the "already decided" shelf, which is
        // indistinguishable from the vote being thrown away — their words: "after
        // approving it vanished". A verdict is only closed when its consequence is
        // visible, so every pick now names the brief that acts on it and that
        // brief's real lane.
        const decided = [];
        const LANE_META = {
          backlog: { label: 'IN BACKLOG — not queued yet', color: '#7E8894' },
          pending: { label: 'QUEUED — waiting for the agent', color: '#5CD1FF' },
          active: { label: 'BEING WORKED RIGHT NOW', color: '#F2D24B' },
          done: { label: 'BUILT — check the LIVE lane on the board', color: '#4FA08B' },
          blocked: { label: 'BLOCKED — Director triaging', color: '#ff6b6b' },
        };
        for (const name of (d.boards ?? [])) {
          const spec = review[name];
          const choice = choices[name];
          const picked = choice?.verdict ?? null;
          const href = '/art/board/' + name;
          if (spec && !picked) {
            // Awaiting the developer's decision — real options, real question.
            const opts = (spec.options ?? []).map(o => ({ id: o.id, label: o.label }));
            opts.push({ id: 'DENY ALL', label: 'DENY ALL' });
            reviews.push({
              name, href, question: spec.question ?? name, color: '#F2D24B', stateLabel: 'AWAITING YOUR EYES',
              picks: opts.map(o => ({
                label: o.label,
                border: o.id === 'DENY ALL' ? '#4d2a2a' : '#24384f',
                bg: 'transparent',
                color: o.id === 'DENY ALL' ? '#ff9d9d' : '#8aa0b6',
                act: () => { if (o.id === 'DENY ALL') { const r = window.prompt('Why deny? Your words guide the next round:'); if (r !== null) this.choose(name, 'DENY ALL', r); } else { this.choose(name, o.id); } },
              })),
            });
          } else if (picked) {
            // DECIDED — surfaced with its consequence, not filed away as history.
            const denied = picked === 'DENY ALL';
            const briefs = (work[name] ?? []).map(w => {
              const m = LANE_META[w.lane] ?? { label: w.lane.toUpperCase(), color: '#8A94A2' };
              return { file: w.file.replace(/\.md$/, ''), laneLabel: m.label,
                color: m.color, bg: m.color + '1c', border: m.color + '55' };
            });
            // Never claim work is under way when nothing references the board. An
            // honest "nothing queued yet" is the whole reason this section exists.
            const live = briefs.filter(b => b.laneLabel !== LANE_META.backlog.label);
            const statusLine = briefs.length === 0
              ? (denied ? 'No regeneration round queued yet — the Director owes you one.'
                        : 'Nothing is queued against this pick yet — the Director owes you a brief.')
              : live.length === 0
                ? 'The brief that builds this exists but is not queued yet.'
                : (denied ? 'A new round is in the queue, carrying your reason verbatim.'
                          : 'The brief that builds your pick:');
            decided.push({
              name, href,
              verdictLabel: denied
                ? ('✗ YOU DENIED — "' + (choice.reason ?? 'new round requested') + '"')
                : ('✓ YOU PICKED ' + picked + ' — ' + name),
              when: choice.at ? new Date(choice.at).toLocaleString() : '',
              accent: denied ? '#ff6b6b' : '#7fe0c4',
              statusLine,
              statusColor: briefs.length === 0 ? '#F2D24B' : '#9fb2c6',
              briefs, hasBriefs: briefs.length > 0,
            });
          } else {
            // Ratified reference — decided long ago, kept for the record.
            const stateLabel = '✓ RATIFIED';
            refs.push({
              name, href, stateLabel,
              open: this.__openRef === name,
              toggleLabel: this.__openRef === name ? 'HIDE' : 'VIEW',
              toggle: () => { this.__openRef = this.__openRef === name ? null : name; this.setState({}); },
            });
          }
        }
        return { reviews, refs, decided,
          hasRefs: refs.length > 0, hasDecided: decided.length > 0,
          empty: reviews.length === 0 && refs.length === 0 && decided.length === 0 };
      }
    }
